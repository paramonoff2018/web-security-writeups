# env-template — шаблон переменных окружения

> ⚠️ Скопируй в файл `.env` (он в `.gitignore`) и заполни реальными значениями.
> НИКОГДА не коммить `.env` в репозиторий (public GitHub!).
> Имя `.env.example` система безопасности IDE блокирует, поэтому шаблон здесь.

```bash
# === Идентификация хантера ===
HUNTER_HEADER="X-Bug-Bounty: Al_x_Pv"

# === Модель (KodikRouter) ===
KODIK_API_KEY=...
KODIK_BASE_URL=https://api.kodikrouter.ru/v1
DEEPSEEK_PRO_MODEL=...
DEEPSEEK_FLASH_MODEL=...

# === Поиск в интернете (Brave MCP) ===
BRAVE_API_KEY=...

# === Тестовые аккаунты Avito (роли) ===
ACCOUNT_USER1=...
ACCOUNT_USER2=...
ACCOUNT_SELLER=...
ACCOUNT_ADMIN=...

# === Токены (после логина) ===
AVITO_SESSION_COOKIE=...
AVITO_ACCESS_TOKEN=...
AVITO_CSRF_TOKEN=...

# === Kali (SSH для Remote-SSH / агента) ===
KALI_SSH_HOST=192.168.56.101
KALI_SSH_USER=...
KALI_SSH_KEY_PATH=...

# === Burp proxy ===
BURP_PROXY=http://127.0.0.1:8080
```
